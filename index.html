use actix_web::{web, App, HttpServer, Responder};

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    HttpServer::new(|| {
        App::new()
            .route("/health", web::get().to(health))
            .route("/api/v1/data", web::get().to(get_data))
    })
    .bind("0.0.0.0:8080")?
    .run()
    .await
}

async fn health() -> impl Responder {
    "OK"
